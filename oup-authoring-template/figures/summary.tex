% RIsearch + RIOT overview — TikZ source
% Stacked workflow: RIsearch search core (top, blue) and the RIOT
% off-target pipeline (bottom, green); dashed arrow = RIOT calls RIsearch
% in-process via the PyO3 bindings.
% \usetikzlibrary{arrows.meta,positioning,backgrounds,fit,calc} loaded in main.tex.

% ── Palette ──────────────────────────────────────────────────────────────────
\definecolor{CoreBlue}{RGB}{197,217,241}     % RIsearch container
\definecolor{CoreBorder}{RGB}{68,114,196}
\definecolor{PipeGreen}{RGB}{210,229,191}     % RIOT container
\definecolor{PipeBorder}{RGB}{83,153,51}

\begin{tikzpicture}[
  font=\sffamily,
  >=Stealth,
  sbox/.style={
    draw=black!55, fill=white, rounded corners=3pt, align=center,
    inner sep=4pt, minimum height=1.05cm, text width=2.5cm,
    font=\sffamily\scriptsize,
  },
  io/.style={
    draw=black!45, fill=black!6, rounded corners=2pt, align=center,
    inner sep=3pt, minimum height=1.05cm, text width=1.8cm,
    font=\sffamily\scriptsize,
  },
  flow/.style={->, thick, draw=black!60},
  call/.style={->, thick, dashed, draw=PipeBorder, line width=1pt},
  ttl/.style={font=\sffamily\bfseries\normalsize, anchor=west},
]

% ── RIsearch row (top) ───────────────────────────────────────────────────────
\node[io]   (tgt)  at (0,0)            {Target\\FASTA};
\node[sbox] (idx)  [right=0.55cm of tgt]  {Suffix-array\\index};
\node[sbox] (srch) [right=0.45cm of idx]  {Seed-and-extend\\DP search};
\node[io]   (hits) [right=0.55cm of srch] {Interaction\\hits (TSV)};

\draw[flow] (tgt)  -- (idx);
\draw[flow] (idx)  -- (srch);
\draw[flow] (srch) -- (hits);

\node[ttl] (ristitle) at (tgt.west |- 0,1.05) {RIsearch \textemdash\ search core};

\begin{scope}[on background layer]
  \node[fit=(ristitle)(tgt)(hits),
        draw=CoreBorder, fill=CoreBlue, rounded corners=8pt,
        inner sep=9pt, line width=1pt] (risbox) {};
\end{scope}

% ── RIOT row (bottom) ────────────────────────────────────────────────────────
\node[sbox] (s1) at (0,-3.5)           {(\textit{i})\\RIsearch\\index \& search\\(PyO3)};
\node[sbox] (s2) [right=0.45cm of s1]  {(\textit{ii})\\Intersection\\(NCLS index)};
\node[sbox] (s3) [right=0.45cm of s2]  {(\textit{iii})\\Accessibility\\(ViennaRNA API)};
\node[sbox] (s4) [right=0.45cm of s3]  {(\textit{iv})\\Partition function\\$\to P_{\mathrm{off}}$};
\node[io]   (out) [right=0.55cm of s4] {Off-target\\probs (TSV)};

\draw[flow] (s1) -- (s2);
\draw[flow] (s2) -- (s3);
\draw[flow] (s3) -- (s4);
\draw[flow] (s4) -- (out);

\node[ttl] (riottitle) at (s1.west |- 0,-2.2)
  {RIOT \textemdash\ siRNA off-target pipeline \;(\texttt{pip install riot})};

\begin{scope}[on background layer]
  \node[fit=(riottitle)(s1)(out),
        draw=PipeBorder, fill=PipeGreen, rounded corners=8pt,
        inner sep=9pt, line width=1pt] (riotbox) {};
\end{scope}

% ── Dashed in-process call: RIOT stage (i) -> RIsearch ───────────────────────
\draw[call] (s1.north) -- (s1.north |- risbox.south)
  node[midway, right=1pt, font=\sffamily\scriptsize, text=PipeBorder!90!black]
  {in-process via PyO3 bindings};

\end{tikzpicture}
