% ── Palette ──────────────────────────────────────────────────────────────────
\definecolor{StepBlue}{RGB}{186,214,235}    % light blue fill  (steps)
\definecolor{StepBorder}{RGB}{68,114,196}   % blue border
\definecolor{ToolGreen}{RGB}{168,208,141}   % light green fill (software / API)
\definecolor{ToolBorder}{RGB}{83,153,51}    % green border
\definecolor{ParamGray}{RGB}{217,217,217}   % light gray fill  (params / versions)
\definecolor{ParamBorder}{RGB}{100,100,100} % gray border
\definecolor{IOGray}{RGB}{217,217,217}      % same gray for input
\definecolor{OutOrange}{RGB}{244,177,131}   % orange fill      (output)
\definecolor{OutBorder}{RGB}{180,100,30}    % orange border
\definecolor{PipeTeal}{RGB}{32,130,118}     % teal for pipeline enclosure

\begin{tikzpicture}[
  >=Stealth,
  font=\sffamily\small,
  % Processing step — blue, rounded
  step/.style={
    draw=StepBorder, fill=StepBlue,
    rounded corners=8pt,
    minimum width=2.7cm, minimum height=1.55cm,
    text width=2.5cm, align=center, inner sep=5pt,
  },
  % Software / API call — green, rounded
  tool/.style={
    draw=ToolBorder, fill=ToolGreen,
    rounded corners=6pt,
    minimum width=2.5cm,
    text width=2.3cm, align=center, inner sep=4pt,
    font=\sffamily\scriptsize,
  },
  % Version / parameter annotation — gray, square
  param/.style={
    draw=ParamBorder, fill=ParamGray,
    minimum width=2.3cm, minimum height=0.6cm,
    text width=2.1cm, align=center, inner sep=3pt,
    font=\sffamily\scriptsize,
  },
  % Input box — gray, square
  inbox/.style={
    draw=ParamBorder, fill=IOGray,
    minimum width=1.85cm, minimum height=1.55cm,
    text width=1.65cm, align=center, inner sep=4pt,
  },
  % Output box — orange, square
  outbox/.style={
    draw=OutBorder, fill=OutOrange,
    minimum width=1.85cm, minimum height=1.55cm,
    text width=1.65cm, align=center, inner sep=4pt,
  },
  % Main horizontal flow arrows
  marrow/.style={->, thick, draw=black!70},
  % Tool/param → step arrows (thin, colored)
  tarrow/.style={->, thin, draw=ToolBorder},
  parrow/.style={->, thin, draw=ParamBorder},
]
