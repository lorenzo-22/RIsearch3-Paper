% RIsearch3 system workflow — TikZ source

% ── Palette ──────────────────────────────────────────────────────────────────
\definecolor{StepBlue}{RGB}{179,208,232}    % light blue  — main component boxes
\definecolor{StepBorder}{RGB}{68,114,196}   % blue border
\definecolor{ToolGreen}{RGB}{168,211,141}   % light green — software / API boxes
\definecolor{ToolBorder}{RGB}{83,153,51}    % green border
\definecolor{ParamGray}{RGB}{217,217,217}   % light gray  — parameter / version boxes
\definecolor{ParamBorder}{RGB}{110,110,110} % gray border
\definecolor{OutOrange}{RGB}{244,177,131}   % orange      — output box
\definecolor{OutBorder}{RGB}{190,100,30}    % orange border

\begin{tikzpicture}[
  >=Stealth,
  font=\sffamily\small,
  node distance=0.6cm,
  % ── node styles ──────────────────────────────────────────────────────────
  % Main component boxes — blue, rounded (like "Step N" boxes in reference)
  comp/.style={
    draw=StepBorder, fill=StepBlue,
    rounded corners=8pt,
    minimum width=2.8cm, minimum height=1.6cm,
    text width=2.6cm, align=center, inner sep=5pt,
  },
  % Software / API call boxes — green, rounded (like fastQC/Salmon in reference)
  tool/.style={
    draw=ToolBorder, fill=ToolGreen,
    rounded corners=6pt,
    minimum width=2.6cm, minimum height=0.7cm,
    text width=2.4cm, align=center, inner sep=3pt,
    font=\sffamily\scriptsize,
  },
  % Parameter / version boxes — gray, square (like "fastQC v.0.11.9" in reference)
  param/.style={
    draw=ParamBorder, fill=ParamGray,
    minimum width=2.6cm, minimum height=0.6cm,
    text width=2.4cm, align=center, inner sep=3pt,
    font=\sffamily\scriptsize,
  },
  % Input box — gray, square
  inbox/.style={
    draw=ParamBorder, fill=ParamGray,
    minimum width=1.9cm, minimum height=1.6cm,
    text width=1.7cm, align=center, inner sep=4pt,
  },
  % Output box — orange, square
  outbox/.style={
    draw=OutBorder, fill=OutOrange,
    minimum width=1.9cm, minimum height=1.6cm,
    text width=1.7cm, align=center, inner sep=4pt,
  },
  % Arrows
  marrow/.style={->, thick, draw=black!65},
  tarrow/.style={->, thin, draw=ToolBorder},
  parrow/.style={->, thin, draw=ParamBorder},
]

% ── MAIN ROW ─────────────────────────────────────────────────────────────────

\node[inbox] (input) {
  Query siRNAs\\(FASTA)\\[4pt]
  Target\\transcriptome\\(FASTA)
};

\node[comp, right=0.55cm of input] (core) {
  \textbf{RIsearch3 Core}\\[3pt]
  {\footnotesize\itshape Rust}
};

\node[comp, right=0.55cm of core] (py) {
  \textbf{Python Bindings}\\[3pt]
  {\footnotesize\itshape PyO3\,$\cdot$\,Maturin}
};

\node[comp, right=0.55cm of py] (pipe) {
  \textbf{Off-Target Pipeline}\\[3pt]
  {\footnotesize\itshape Python\,$\cdot$\,Polars}
};

\node[outbox, right=0.55cm of pipe] (output) {
  $p_{\mathrm{off}}$\\per transcript\\[4pt]
  $P_{\mathrm{OFF}}$\\aggregate\\[4pt]

};

% ── TOOL NODES — below each component ────────────────────────────────────────

\node[tool, below=0.65cm of core] (t-core) {
  \texttt{risearch index}\\[1pt]
  \texttt{risearch search}
};

\node[tool, below=0.65cm of py] (t-py) {
  \texttt{risearch.index()}\\[1pt]

};

\node[tool, below=0.65cm of pipe] (t-pipe) {
